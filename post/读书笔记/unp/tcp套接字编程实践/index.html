本文以TCP Echo客户端与服务器为例，系统讲解套接字编程中的启动、连接、正常终止、异常终止及异常恢复过程。

内容涵盖三路握手、四次挥手、TIME_WAIT、SIGCHLD 与僵死进程、wait/waitpid 的差异，以及 EINTR、ECONNABORTED、EPIPE、ETIMEDOUT 和 ECONNRESET 等错误处理。

同时分析服务器终止、RST、SIGPIPE、主机崩溃或重启等网络边界条件，并说明 select、SO_KEEPALIVE、SO_LINGER 等机制的作用。通过实际命令、进程状态和代码示例，文章总结了可靠多进程服务器与客户端应具备的信号处理、系统调用重试和套接字异常处理能力。

服务器正常启动和关闭

TCP Echo服务器

#include <errno.h>
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <strings.h>
#include <sys/socket.h>
#include <unistd.h>
#define SERV_PORT 9877
#define LISTENQ   9
#define MAXLINE   1024

ssize_t writen(int fd, const void* vptr, size_t n)
{
    size_t      nleft;
    ssize_t     nwritten;
    const char* ptr;

    ptr   = vptr;
    nleft = n;
    while (nleft > 0)
    {
        if ((nwritten = write(fd, ptr, nleft)) <= 0)
        {
            if (nwritten < 0 && errno == EINTR)
            {
                continue;
            }
            else
            {
                // 其它错误：如果已经写入部分数据，返回已写入；若尚未写入则返回 -1
                return (nleft == n) ? -1 : (ssize_t)(n - nleft);
            }
        }
        nleft -= nwritten;
        ptr   += nwritten;
    }
    return (nleft == n) ? -1 : (ssize_t)(n - nleft);
}
static void str_echo(int sockfd)
{
    ssize_t n;
    char    buf[MAXLINE];

again:
    while ((n = read(sockfd, buf, MAXLINE)) > 0)
    {
        writen(sockfd, buf, n);
    }

    if (n < 0 && errno == EINTR)
    {
        goto again;
    }
    else if (n < 0)
    {
        perror("str_echo: read error");
    }
}

int main(int argc, char** argv)
{
    int                listenfd, connfd;
    pid_t              childpid;
    socklen_t          clilen;
    struct sockaddr_in cliaddr, servaddr;

    // Create socket
    if ((listenfd = socket(AF_INET, SOCK_STREAM, 0)) < 0)
    {
        perror("socket creation failed");
        exit(EXIT_FAILURE);
    }

    bzero(&servaddr, sizeof(servaddr));

    // Configure server address
    servaddr.sin_family      = AF_INET;
    servaddr.sin_addr.s_addr = htonl(INADDR_ANY);
    servaddr.sin_port        = htons(SERV_PORT);

    // Bind socket to address
    if (bind(listenfd, (struct sockaddr*)&servaddr, sizeof(servaddr)) < 0)
    {
        perror("bind failed");
        exit(EXIT_FAILURE);
    }

    // Start listening for connections
    if (listen(listenfd, LISTENQ) < 0)
    {
        perror("listen failed");
        exit(EXIT_FAILURE);
    }

    for (;;)
    {
        clilen = sizeof(cliaddr);
        // Accept incoming connection
        if ((connfd = accept(listenfd, (struct sockaddr*)&cliaddr, &clilen)) < 0)
        {
            perror("accept failed");
            continue;  // Continue to accept next connection
        }

        // Fork new process to handle connection
        if ((childpid = fork()) == -1)
        {
            perror("fork failed");
            close(connfd);
            continue;
        }
        else if (childpid == 0)  // Child process
        {
            close(listenfd);
            // handle connection here
            str_echo(connfd);
            exit(EXIT_SUCCESS);
        }
        close(connfd);  // Parent closes connected socket
    }
}

客户端示例

#include <arpa/inet.h>
#include <errno.h>
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <strings.h>
#include <sys/socket.h>
#include <unistd.h>
#define SERV_PORT 9877
#define MAXLINE   1024

ssize_t readline(int fd, void* vptr, size_t maxlen)
{
    ssize_t n, rc;
    char    c, *ptr;
    ptr = vptr;
    for (n = 1; n < maxlen; n++)
    {
        if ((rc = read(fd, &c, 1)) == 1)
        {
            *ptr++ = c;
            if (c == '\n')
            {
                break;
            }
        }
        else if (rc == 0)
        {
            /* EOF */
            *ptr = 0;
            return (n - 1);
        }
        else
        {
            if (errno == EINTR)
            {
                continue;
            }
            return -1;
        }
    }
    *ptr = 0;
    return n;
}
ssize_t writen(int fd, const void* vptr, size_t n)
{
    size_t      nleft;
    ssize_t     nwritten;
    const char* ptr;

    ptr   = vptr;
    nleft = n;
    while (nleft > 0)
    {
        if ((nwritten = write(fd, ptr, nleft)) <= 0)
        {
            if (nwritten < 0 && errno == EINTR)
            {
                continue;
            }
            else
            {
                // 其它错误：如果已经写入部分数据，返回已写入；若尚未写入则返回 -1
                return (nleft == n) ? -1 : (ssize_t)(n - nleft);
            }
        }
        nleft -= nwritten;
        ptr   += nwritten;
    }
    return (nleft == n) ? -1 : (ssize_t)(n - nleft);
}
static void str_cli(FILE* fp, int sockfd)
{
    char sendline[MAXLINE], recvline[MAXLINE];
    while (fgets(sendline, MAXLINE, fp) != NULL)
    {
        writen(sockfd, sendline, strlen(sendline));
        if (readline(sockfd, recvline, MAXLINE) == 0)
        {
            perror("str_cli: server terminated prematurely");
            exit(0);
        }
        fputs(recvline, stdout);
    }
}

int main(int argc, char** argv)
{
    int                sockfd;
    struct sockaddr_in servaddr;

    if (argc != 2)
    {
        perror("usage: tcpcli <IPaddresses>");
        exit(-1);
    }

    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0)
    {
        perror("socket error");
        exit(-1);
    }

    bzero(&servaddr, sizeof(servaddr));

    servaddr.sin_family = AF_INET;
    servaddr.sin_port   = htons(SERV_PORT);
    if (inet_pton(AF_INET, argv[1], &servaddr.sin_addr) <= 0)
    {
        perror("inet_pton error");
        exit(-1);
    }

    if (connect(sockfd, (struct sockaddr*)&servaddr, sizeof(servaddr)) < 0)
    {
        perror("connect error");
        exit(-1);
    }

    str_cli(stdin, sockfd);

    exit(0);
}

正常启动

首先启动服务器，命令如下所示：

$ ./tcpserver01 &
[1] 1960588
$ netstat -a | grep 9877
tcp        0      0 0.0.0.0:9877            0.0.0.0:*               LISTEN
$ netstat -a
Active Internet connections (servers and established)
Proto Recv-Q Send-Q Local Address           Foreign Address         State
tcp        0      0 0.0.0.0:9877            0.0.0.0:*               LISTEN

其中0.0.0.0表示INADDR_ANY通配地址，并监听9877端口。

随后调用./tcpcli01 127.0.0.1启动客户端，此时客户端调用socket和connect，后者会触发TCP的三路握手过程。当三路握手完成后，客户端中的connect和服务器中的accept均会返回，连接建立成功。

  1. 此时客户端调用str_cli函数，并阻塞于fgets函数调用，等待用户输入。
  2. 服务器的accept函数返回后，服务器调用fork，再由子进程调用str_echo。该函数调用read等待客户端传入文本而阻塞。
  3. 同时服务器父进程再次调用accept函数并阻塞，等待下一个客户端连接。

此时有3个睡眠的进程分别是客户端进程、服务器父进程和服务器子进程。如下所示：

$ netstat -a
Active Internet connections (servers and established)
Proto Recv-Q Send-Q Local Address           Foreign Address         State
tcp        0      0 0.0.0.0:9877            0.0.0.0:*               LISTEN
tcp        0      0 localhost:35418         localhost:9877          ESTABLISHED
tcp        0      0 localhost:9877          localhost:35418         ESTABLISHED

其中第1行是服务器父进程，处于监听端口状态，等待下一个客户端连接；第2行是客户端子进程，因为本地端口号是35418；第3行是服务器子进程。

使用ps检查这些进程的状态和关系：

$ ps  -ao pid,ppid,tty,stat,args,wchan
    PID    PPID TT       STAT COMMAND                     WCHAN
1960588 1960230 pts/4    S    ./tcpserver01               inet_csk_accept
1965854 1960230 pts/4    S+   ./tcpcli01 127.0.0.1        wait_woken
1965855 1960588 pts/4    S    ./tcpserver01               wait_woken

其中PID和PPID给出了进程间的关系，第3行是服务器子进程。

3个进程都处于睡眠状态，同时WCHAN列出了相应的条件。inet_csk_accept表示进程阻塞于accept。wait_woken表示进程阻塞于I/O操作上。

正常终止

至此连接已经建立，不管客户端输入什么，都会回射到它的标准输出中。

$ ./tcpcli01 127.0.0.1
hello,world
hello,world
good bye
good bye

当键入终端EOF字符（Ctrl-D）以终止客户端，此时可以看到

$ netstat -a | grep 9877
tcp        0      0 0.0.0.0:9877            0.0.0.0:*               LISTEN
tcp        0      0 localhost:35418         localhost:9877          TIME_WAIT

客户端进入了TIME_WAIT状态，监听服务器仍旧在等待下一个客户端的连接。

具体步骤如下：

  1. 当键入EOF字符后，fgets函数返回空指针，然后str_cli函数返回。
  2. 当str_cli函数返回到main函数后，main函数通过exit终止。
  3. 客户端进程终止时会关闭所有描述符，因此当客户端打开的套接字由内核关闭时，会导致客户端TCP发送FIN给服务器，服务器TCP则回复ACK，这是TCP连接终止序列的前半部分。
  4. 当服务器TCP收到FIN之前，服务器子进程还阻塞在read函数，收到FIN后则返回0。这导致str_echo函数返回服务器子进程的main函数。
  5. 服务器子进程通过调用exit终止。
  6. 服务器子进程打开的所有描述符随之关闭。由子进程关闭已连接套接字会引发TCP连接终止序列的最后两个分节：一个从服务器到客户端的FIN和一个从客户端到服务器的ACK。
  7. 至此，连接完全终止，客户端套接字进入TIME_WAIT状态。
  8. 在服务器子进程终止时，会给父进程发送一个SIGCHLD信号。但是上面的示例中没有捕获该信号，因此子进程进入僵死状态。
$ ps  -ao pid,ppid,tty,stat,args,wchan
    PID    PPID TT       STAT COMMAND                     WCHAN
1960588 1960230 pts/4    S    ./tcpserver01               inet_csk_accept
1965855 1960588 pts/4    Z    [tcpserver01] <defunct>     -
步骤动作服务器状态客户端状态
1客户发送 FINCLOSE_WAITFIN_WAIT_1
2服务器 ACK FINCLOSE_WAITFIN_WAIT_2
3服务器发送 FIN（关闭 connfd 时）LAST_ACKFIN_WAIT_2
4客户 ACK FINCLOSEDTIME_WAIT

两个方向的 FIN 都是进程退出时内核自动发送的，不需要应用层代码干预。

如果服务器发送了FIN，但客户端的 ACK 始终未到达（如客户崩溃或网络中断），内核会持续重传 FIN，直到重传超时（约 2 分钟）后放弃，强制进入 CLOSED 状态，此时进程资源才被回收。

信号与僵死进程

信号是告知某个进程发生了某个事件的通知，也称为软件中断。

信号可以：

  1. 由一个进程发送给另一个进程。
  2. 由内核发送给某个进程。

每个信号都有一个与之关联的处置，可以通过sigaction函数来设置一个信号的处置。

有3种信号处置的选择：

  1. 设置一个函数，当特定信号发生时该函数就被调用。函数称为信号处理函数，这种行为称为捕获信号。SIGKILL和SIGSTOP不能被捕获。
  2. 设置信号处置为SIG_IGN，用于忽略某个信号。其中SIGKILL和SIGSTOP不能被忽略。
  3. 设置信号处置为SIG_DFL，使用信号的默认处置。个别信号的默认处置是忽略，例如SIGCHLD和SIGURG。

建立信号处置的方法是调用sigaction函数或signal函数（第2个参数可以设置为函数指针或者常值SIG_IGN或SIG_DFL），函数原型如下：

#include <signal.h>
typedef void (*sighandler_t)(int);
struct sigaction
{
    void (*sa_handler)(int);
    void (*sa_sigaction)(int, siginfo_t *, void *);
    sigset_t sa_mask;
    int sa_flags;
    void (*sa_restorer)(void);
};
int sigaction(int signum, const struct sigaction *act, struct sigaction *oldact);
sighandler_t signal(int signum, sighandler_t handler);
成员含义和作用
sa_handler简单处理函数，形式为 void handler(int signum)。也可设为 SIG_DFL（默认处理）或 SIG_IGN（忽略信号）。
sa_sigaction扩展处理函数，形式为 void handler(int signum, siginfo_t *info, void *ucontext)，可获取信号来源等额外信息。通常只有设置 SA_SIGINFO 标志后才使用它。
sa_mask信号处理函数运行期间额外屏蔽的信号集合。处理函数执行时，触发它的信号通常也会自动屏蔽；处理结束后恢复。
sa_flags配置处理行为的标志，例如 SA_SIGINFO 使用扩展处理函数，SA_RESTART 尽量让被信号中断的系统调用自动重启。
sa_restorer某些系统或 ABI 内部使用的恢复函数指针，属于实现细节；应用程序通常不应设置或直接使用。

sa_handler 和 sa_sigaction 是两种互斥的处理函数形式，不是同时生效的两个回调。典型用法是设定 sa_flags，然后按所选形式赋值。

POSIX信号语义

POSIX信号处理规则：

  1. 一旦安装了信号处理函数，它便一直安装着。
  2. 在一个信号处理函数运行期间，正被递交的信号是阻塞的。并且，安装处理函数时在传递给sigaction函数的sa_mask信号集中指定的任何额外信号也被阻塞。如果sa_mask设置为空集，则意味着除了被捕获的信号，没有额外信号被阻塞。
  3. 如果一个信号在被阻塞期间产生了一次或多次，那么该信号被解阻塞之后通常只递交一次，也就是说UNIX信号默认是不排队的。
  4. 使用sigprocmask函数可以选择性地阻塞和解阻塞一组信号。这使得在一段临界区代码执行期间，防止捕获某些信号，保护这段代码运行成为可能。

僵死进程

在子进程终止时会给父进程发送一个SIGCHLD信号，如果父进程未捕获这个信号进行回收子进程，那么子进程就会进入僵死状态。

设置僵死状态的目的是维护子进程的信息，以便父进程在以后某个时候获取。主要包括子进程的进程ID、终止状态以及资源利用信息（CPU时间、内存使用量等）。

如果一个进程终止，而该进程有子进程处于僵死状态，那么它的所有僵死子进程的父进程ID将被重置为1（init进程）。继承这些子进程的init进程则会清理这些僵死子进程。

如果进程不处理其僵死子进程，它们就会占用内核的空间，最终可能耗尽进程资源。

如何处理僵死子进程呢？只需要在收到SIGCHLD信号之后回收子进程的资源即可，具体做法则是1创建SIGCHLD信号函数，2在信号处理函数中调用wait回收子进程。

void sig_chld(int signo)
{
    pid_t pid;
    int   stat;
    pid = wait(&stat);
    printf("child %d terminated", pid);
    return;
}

当SIGCHLD信号被递交时，父进程阻塞在accept调用。既然该信号是由父进程阻塞于慢系统调用时捕获的，内核就会使accept返回EINTR错误（被中断的慢系统调用）。然而此时父进程还没有处理该错误，于是会终止。

针对于这种慢系统调用的基本处理规则是：当阻塞于某个慢系统调用的进程捕获某个信号且相应信号处理函数返回时，该系统调用可能会返回一个EINTR错误。有些内核自动重启某些被中断的系统调用，不过为了便于移植，必须处理慢系统调用的EINTR错误。处理方式一般是重新拉起对应的慢系统调用函数。

但是有个函数不能重启：connect。如果该函数返回EINTR，不能再次调用来重启，而应该使用select来等待连接完成。这是因为调用connect之后，内核已经发起连接，如果重启则会重复发起连接直接导致错误。

因此对于上面的服务器函数的处理方式如下：

for (;;)
{
    clilen = sizeof(cliaddr);
    if ((connfd = accept(listenfd, (struct sockaddr*)&cliaddr, &clilen)) < 0)
    {
        if (errno == EINTR)
        {
            continue;
        }
        else
        {
            perror("accept error");
        }
    }
    // ...
}

wait和waitpid函数

#include <sys/wait.h>
pid_t wait(int* wstatus);
pid_t waitpid(pid_t pid, int* wstatus, int options);

// 返回值：成功返回进程ID，出错返回0或-1，当指定WNOHANG且没有进程终止则waitpid返回0，其他情况返回-1

这两个函数均可以返回终止的子进程ID和通过wstatus返回子进程终止状态，可以通过wtatus判断子进程是正常终止、由信号杀死还是由作业控制停止等。

如果调用wait的进程没有已终止的子进程，但是有正在运行的一个或多个子进程，那么wait函数会阻塞直到有子进程终止为止。

waitpid函数可以控制等待哪个子进程和是否阻塞。pid参数用于指定要等待的子进程，-1表示等待第一个终止的子进程。options参数运行指定附加选项，WNOHANG用于告知内核没有已终止子进程时不要阻塞。

那么在使用上这两个函数有什么区别呢？

看如下示例，客户端进程创建了5个和服务器的连接

// 客户端进程
int main(int argc, char* argv[])
{
    int                sockfd[5];
    int                i;
    struct sockaddr_in servaddr;

    if (argc != 2)
    {
        printf("usage: tcpcli <ipaddress>");
        return -1;
    }

    for (i = 0; i < 5; i++)
    {
        sockfd[i] = socket(AF_INET, SOCK_STREAM, 0);

        bzero(&servaddr, sizeof(servaddr));
        servaddr.sin_family = AF_INET;
        servaddr.sin_port   = htons(SERV_PORT);
        inet_pton(AF_INET, argv[1], &servaddr.sin_addr);

        if (connect(sockfd[i], (struct sockaddr*)&servaddr, sizeof(servaddr)) != 0)
        {
            perror("connect error");
            return -1;
        }
    }

    str_cli(stdin, sockfd[0]);
    exit(0);
}

// 服务器进程
void sig_chld(int signo)
{
    pid_t pid;
    int   stat;
    pid = wait(&stat);
    printf("child %d terminated", pid);
    return;
}
int main(int argc, char* argv[])
{
    int                listenfd, connfd;
    pid_t              childpid;
    struct sockaddr_in cliaddr, servaddr;
    socklen_t          clilen;

    listenfd = socket(AF_INET, SOCK_STREAM, 0);
    if (listenfd < 0)
    {
        perror("socket error");
        exit(-1);
    }

    bzero(&servaddr, sizeof(struct sockaddr_in));
    servaddr.sin_family      = AF_INET;
    servaddr.sin_port        = htons(SERV_PORT);
    servaddr.sin_addr.s_addr = htonl(INADDR_ANY);

    if (bind(listenfd, (struct sockaddr*)&servaddr, sizeof(servaddr)) < 0)
    {
        perror("bind error");
        close(listenfd);
        exit(-1);
    }

    if (listen(listenfd, LISTENQ) < 0)
    {
        perror("listen error");
        close(listenfd);
        exit(-1);
    }

    signal(SIGCHLD, sig_chld);

    for (;;)
    {
        clilen = sizeof(cliaddr);
        if ((connfd = accept(listenfd, (struct sockaddr*)&cliaddr, &clilen)) < 0)
        {
            if (errno == EINTR)
            {
                continue;
            }
            else
            {
                perror("accept error");
            }
        }
        char buf[80] = {0};
        printf(
            "connect from %s:%d\n", inet_ntop(AF_INET, &cliaddr.sin_addr, buf, sizeof(buf)), ntohs(cliaddr.sin_port));
        if ((childpid = fork()) == 0)
        {
            close(listenfd);
            str_echo(connfd);
            exit(0);
        }
        close(connfd);
    }
}

当客户端终止时，所有打开的描述符由内核关闭，所有5个连接同时终止。此时会引发5个FIN，会导致服务器的5个子进程同时终止。这又会导致5个SIGCHLD信号同时递交给父进程。

这种信号同时递交会引发问题，当客户端进程退出时，服务器的5个子进程也同时终止，但是仍然有3个子进程作为僵死进程存在。

$ ps  -ao pid,ppid,tty,stat,args,wchan
    PID    PPID TT       STAT COMMAND                     WCHAN
1768614 1588584 pts/5    S+   ./tcpserv01                 inet_csk_accept
1769218 1768614 pts/5    Z+   [tcpserv01] <defunct>       -
1769219 1768614 pts/5    Z+   [tcpserv01] <defunct>       -
1769220 1768614 pts/5    Z+   [tcpserv01] <defunct>       -

由此可见，建立一个信号处理函数并调用wait并不足以防止出现僵死进程。原因在于当出现第1个SIGCHLD信号时，系统执行了信号处理函数，而在信号处理函数执行期间出现的4个SIGCHLD信号会被阻塞，等待信号处理函数执行完之后只递交一次（符合POSIX的信号处理语义）。

针对于这种问题的处理方法是调用waitpid而非wait，即在一个循环内调用waitpid，以获取所有已终止子进程的状态，同时必须指定WNOHANG选项，在有尚未终止的子进程时不要阻塞。

修复之后服务器最终代码如下：

void sig_chld(int signo)
{
    pid_t pid;
    int   stat;
    while ((pid = waitpid(-1, &stat, WNOHANG)) > 0)
    {
        printf("child %d terminated", pid);
    }
    return;
}
int main(int argc, char* argv[])
{
    int                listenfd, connfd;
    pid_t              childpid;
    struct sockaddr_in cliaddr, servaddr;
    socklen_t          clilen;

    listenfd = socket(AF_INET, SOCK_STREAM, 0);
    if (listenfd < 0)
    {
        perror("socket error");
        exit(-1);
    }

    bzero(&servaddr, sizeof(struct sockaddr_in));
    servaddr.sin_family      = AF_INET;
    servaddr.sin_port        = htons(SERV_PORT);
    servaddr.sin_addr.s_addr = htonl(INADDR_ANY);

    if (bind(listenfd, (struct sockaddr*)&servaddr, sizeof(servaddr)) < 0)
    {
        perror("bind error");
        close(listenfd);
        exit(-1);
    }

    if (listen(listenfd, LISTENQ) < 0)
    {
        perror("listen error");
        close(listenfd);
        exit(-1);
    }

    signal(SIGCHLD, sig_chld);

    for (;;)
    {
        clilen = sizeof(cliaddr);
        if ((connfd = accept(listenfd, (struct sockaddr*)&cliaddr, &clilen)) < 0)
        {
            if (errno == EINTR)
            {
                continue;
            }
            else
            {
                perror("accept error");
            }
        }
        char buf[80] = {0};
        printf(
            "connect from %s:%d\n", inet_ntop(AF_INET, &cliaddr.sin_addr, buf, sizeof(buf)), ntohs(cliaddr.sin_port));
        if ((childpid = fork()) == 0)
        {
            close(listenfd);
            str_echo(connfd);
            exit(0);
        }
        close(connfd);
    }
}

这个服务器版本解决了3个问题：

  1. 当fork子进程时，捕获SIGCHLD信号。
  2. 当捕获信号时，处理了可能被中断的系统调用
  3. SIGCHLD的信号处理函数必须正确编写，使用waitpid可以避免留下僵死子进程。

健壮性与边界条件

accept返回前连接终止

类似于SIGCHLD信号处理中断accept函数调用，还有一种情形会导致accept返回非致命的错误，在这种情况下，只需要再次调用accept。

这种情形就是在三路握手完成从而建立连接之后，客户端TCP却发送了一个RST报文。在服务器端来看，就是该连接已由TCP排队，等待服务器进程调用accept的时候RST到达。

如下图所示情形：

针对于这种情形，POSIX指出accept函数返回-1的同时，errno必须是ECONNABORTED（“software caused error”，软件引起的连接中止）。

这种情形的验证条件如下：

  1. 启动服务器，调用socket、bind、listen，然后在调用accept之前休眠一段时间。
  2. 服务器休眠期间，启动客户端，调用socket、connect，并在connect返回之后，设置SO_LINGER套接字以产生这个RST。
  3. SO_LINGER套接字的作用是控制close()的行为，决定关闭套接字时未发送数据的处理方式。
  4. 服务器所在系统的内核选择把这个已复位的待接受连接从完成队列移除，并将 ECONNABORTED 返回给 accept()。（这一点在Ubuntu24.04上未复现，accept成功返回，但是在后续read阶段报错str_echo: read error: Connection reset by peer ）

SO_LINGER选项通过struct linger结构体配置，如下所示：

struct linger {
    int l_onoff; // 是否开启linger功能
    int l_linger; // 逗留时间(秒)
}

取值和close()的行为如下：

l_onoffl_lingerclose()的行为
0任意默认行为：立即返回，内核在后台继续发送未完成数据，然后正常进行四次挥手
非00强制立即关闭（RST）：丢弃发送缓冲区所有数据，直接发送RST报文终止连接，不进行四次挥手
非0>0逗留关闭：close()阻塞等待，直到未发送数据全部达到对端并收到ACK或者超过l_linger设置的超时时间（超时后转为RST关闭）

SO_LINGER设置示例如下：

int main(int argc, char** argv)
{
    int                sockfd;
    struct sockaddr_in servaddr;

    if (argc != 2)
    {
        perror("usage: tcpcli <IPaddresses>");
        exit(-1);
    }

    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0)
    {
        perror("socket error");
        exit(-1);
    }

    bzero(&servaddr, sizeof(servaddr));

    servaddr.sin_family = AF_INET;
    servaddr.sin_port   = htons(SERV_PORT);
    if (inet_pton(AF_INET, argv[1], &servaddr.sin_addr) <= 0)
    {
        perror("inet_pton error");
        exit(-1);
    }

    if (connect(sockfd, (struct sockaddr*)&servaddr, sizeof(servaddr)) < 0)
    {
        perror("connect error");
        exit(-1);
    }

    struct linger ling;
    // 强制立即关闭连接，发送RST，丢弃缓冲区的内容
    ling.l_onoff  = 1;
    ling.l_linger = 0;
    setsockopt(sockfd, SOL_SOCKET, SO_LINGER, &ling, sizeof(ling));

    close(sockfd);

    exit(0);
}

服务器进程终止

在当前的客户端/服务器例子中，当服务器进程终止时，所有连接的文件描述符都会被关闭，此时服务器TCP会向客户端发送一个FIN，而客户端TCP则响应一个ACK（TCP连接终止的前半部分）。客户端TCP接收来自服务器的FIN并响应一个ACK时，客户端进程可能正阻塞在fgets调用上，等待从终端接收数据。

当键入数据后，客户端str_cli调用writen，然后客户端TCP将数据发送给服务器（客户端TCP接收到FIN只表示服务器进程关闭了连接的服务器端，服务器端不再向其中发送任何数据，并没有告知客户端服务器TCP进程已经终止）。

当服务器TCP接收到来自客户端的数据时，由于先前打开对应套接字的进程已经终止，因此会响应一个RST。

客户端进程在调用writen()后会立即调用readline()，此时会立即返回0（表示EOF，由于之前接收到的FIN）。客户端并未接收到预期的字符串，因此会执行perror("str_cli: server terminated prematurely");输出对应的出错信息，然后退出进程。

会出现该问题的原因在于：当FIN到达客户端TCP套接字时，客户端正阻塞在fgets调用上，而客户端实际上需要应对两个描述符，套接字和用户输入，客户端不应该单纯阻塞在这两个源中某个特定源的输入上，而是应该阻塞在其中任何一个源的输入上。

解决方案则是使用select或poll函数，同时监听多个描述符，当服务器进程的终止一旦发生，客户端立即就能检测到。

验证条件如下：

  1. 在同一台主机上启动服务器和客户端程序，并在客户端上键入一行文本，验证一切正常。
  2. 找到服务器子进程的进程ID，并执行kill -9 <pid>杀死该进程。作为进程终止处理的部分工作，子进程中所有打开的文件描述符都会被关闭。这会导致向客户端发送一个FIN，而客户端TCP则响应一个ACK。
  3. SIGCHLD信号被发送给父进程，并得到正确处理，如前文所述。

SIGPIPE信号

当一个进程向某个已经收到RST的套接字执行写操作时，内核向该进程发送一个SIGPIPE信号。该信号的默认行为是终止进程，因此进程必须捕获它以避免被意外终止。

在上一个例子中，如果忽略readline()函数的返回值，继续向该套接字写入更多数据，此时就会引发内核向客户端进程发送SIGPIPE信号，因为第一次写操作已经导致服务器发送了RST。

static void str_cli(FILE* fp, int sockfd)
{
    char sendline[MAXLINE], recvline[MAXLINE];
    while (fgets(sendline, MAXLINE, fp) != NULL)
    {
        /* 这里做两次write，目的是让第1次writen引发RST，再让第2个writen产生SIGPIPE */
        writen(sockfd, sendline, 1);
        sleep(1);
        writen(sockfd, sendline+1, strlen(sendline)-1);
        if (readline(sockfd, recvline, MAXLINE) == 0)
        {
            perror("str_cli: server terminated prematurely");
            exit(0);
        }
        fputs(recvline, stdout);
    }
}

此时执行结果是：

# 当第2次执行writen时，内核向该进程发送了SIGPIPE信号，并执行了默认行为终止进程
$ ./tcpcli03 127.0.0.1
aaa
aaa
ccc
$

当忽略了SIGPIPE信号后，执行结果如下：

# 可以观察到，readline此时没有读取到预期的字符串，所以正常退出
# 同时writen产生的错误导致errno设置为EPIPE，然后输出对应错误
$ ./tcpcli04 127.0.0.1
aaa
aaa
ccc
str_cli: server terminated prematurely: Broken pipe # 这里是正常逻辑

由此可以判断，不论进程是捕获了该信号并从其信号处理函数返回，还是简单的忽略该信号，写操作都会返回EPIPE错误。

如何处理SIGPIPE信号呢？

  1. 如果没有特殊的事情要做，那直接将信号处理方法设置为SIG_IGN，并假设后续的操作会捕获write返回的EPIPE错误并处理。
  2. 如果需要采取特殊措施（比如日志记录），则需要捕获该信号，并在信号处理函数中处理。
  3. 如果使用多个套接字，该信号的递交无法给出具体套接字出错，因此如果必须知道哪个write出错，那么必须忽略该信号或从信号处理函数返回后，再处理对应write返回的EPIPE。

服务器主机崩溃

模拟方法：

  1. 在不同主机上先启动服务器，然后启动客户端。
  2. 在客户端上键入一行文本确认连接正常。
  3. 在网络上断开服务器主机
  4. 在客户端上键入一行文本。

此时键入的文本会有writen()写入内核，然后由客户端TCP作为数据分节发出，客户端阻塞在readline()等待响应。

由于服务器主机已崩溃，从而对客户端的数据分节根本没有响应，那么所返回的错误是ETIMEDOUT。

如果某个中间路由器判定服务器主机已不可达，从而响应一个destination unreachable的ICMP消息，那么所返回的错误就是EHOSTUNREACH或ENETUNREACH。

由于TCP的重传机制，会导致耗时较长。为了更快检测到这种情况，一般需要给readline()调用设置超时时间。

同时这种情况只有客户端主动向服务器发送数据才会检测到，如果想要不主动发送数据也检测到这种情况，则需要使用SO_KEEPALIVE套接字选项。

服务器主机崩溃后重启

模拟方法：先建立连接->再从网络上断开服务器主机->重启服务器主机->再将其连接到网络中。

如果客户端在服务器崩溃过程中没有发送任何数据，那么不会知道服务器主机已经崩溃（前提没有启用SO_KEEPALIVE选项）。

服务器主机崩溃后重启时，会丢失所有崩溃前的所有TCP连接信息，因此服务器TCP对于所收到的客户端的数据分节会响应一个RST。当客户端收到该RST时，正阻塞在readline()调用，会导致该调用返回ECONNRESET错误。

客户端可以通过SO_KEEPALIVE或其他心跳动作检测服务器状态。

服务器主机重启

类Unix系统关机时，init进程会先给所有进程发送SIGTERM信号（该信号可被捕获），等待一段固定的时间（5~20s），然后给所有仍在运行的进程发送SIGKILL信号（该信号不能被捕获）。这么做的原因在于给所有运行的进程一段时间来执行清除和终止动作。

当服务器子进程终止时，它的所有打开的描述符都被关闭，随后就会发生服务器进程终止时，客户端进程阻塞在fgets()调用的问题。
